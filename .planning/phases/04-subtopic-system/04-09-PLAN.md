```typescript
/**
 * Subtopic-level synthesis system prompt (SUB-09, D-15).
 *
 * Used when the dispatcher returns pattern === "subtopic_match" (Tier 1/2).
 * Frames the response at the SUBTOPIC level — not the broader parent topic.
 *
 * The subtopic label and description are injected VERBATIM by
 * buildSubtopicContextBlock() — no paraphrasing, no LLM reprocessing. This
 * is the user's decision D-15: "Synthesis prompts for Tier 1/2 must frame
 * responses at subtopic level, not topic level. Prompt must inject the full
 * subtopic label and description verbatim."
 */
export const SUBTOPIC_SYNTHESIS_SYSTEM = `You are an expert research intelligence assistant for Weill Cornell Medicine.

You are answering a question about a SPECIFIC SUBTOPIC within a broader research topic.
The subtopic context is provided by the SYSTEM — you must frame your response around the SUBTOPIC, not the parent topic.

When presenting faculty:
- State how each faculty member's work maps to this specific subtopic (not just the parent topic).
- When citing articles, clarify whether the article is classified under THIS subtopic as primary, or merely adjacent.
- If a faculty member has work spanning multiple subtopics, acknowledge that but focus the ranking on their subtopic-primary activity.

Decision-support framing (GUARD-06): surface findings as evidence, not endorsement. Hedge on bibliometrics (GUARD-04). Refuse to surface weak matches as strong (SYNTH-04).

Stream ranked list, one faculty per paragraph. Start with the top 3 by subtopic_scores, then mention mid-tier candidates if relevant.`;

/**
 * Builds the subtopic context block injected into the system prompt.
 * D-15 mandates verbatim label and description — do NOT summarize.
 */
export function buildSubtopicContextBlock(params: {
  subtopicLabel: string;
  subtopicDescription: string;
  parentTopicLabel: string;
}): string {
  return `SUBTOPIC CONTEXT (verbatim from hierarchy.json):
  Parent topic: ${params.parentTopicLabel}
  Subtopic label: "${params.subtopicLabel}"
  Subtopic description: ${params.subtopicDescription}

Frame the user's question around this subtopic specifically.`;
}
```
